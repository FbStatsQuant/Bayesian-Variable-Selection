\begin{algorithm}

{\footnotesize
\begin{algorithmic}[1]

\State complete-linkage tree: start from $p$ singleton clusters; $p-1$ times merge the pair $(a,b)$ of active clusters with the smallest $d(a,b)=\max_{j\in a,l\in b}d_{jl}$; among exactly equal distances take the pair whose union has the lexicographically smallest increasing index list (a proper prefix first); the left child of the new node is the child with the smaller minimum index %@ v3_blocks.h:46-108 (pair order 62-66, choice 76-83, left child 85-86, max rule 94-98)

\begin{algorithm}
\caption{Stage~3: model evaluation, the start law $\pi_0$ with its exact sampler, and the start mode $\tSz$.}\label{alg:start}

\begin{algorithm}
\caption{The Markov moves: one round, and the four kinds of move with their acceptance probabilities.}\label{alg:moves}
{\footnotesize
\begin{algorithmic}[1]
\Require a particle $S$ (with $\abs S\le K$), a temperature $\xi$; $\log\gamma_\xi(S)=\log\gamma_0(S)+\xi\Delta(S)$ from \textsc{Eval}; the graph $\mathcal N$; $B$; $m:=p+B+2$
\Ensure $S$ updated in place; each move leaves $\pi_\xi\propto\gamma_0e^{\xi\Delta}$ invariant
\Procedure{Round}{$S,\xi$}
\State draw a uniformly random order of the $m$ labels $G_1,\dots,G_p,U,N_1,\dots,N_B,I$ by Fisher--Yates ($m-1$ uniforms); \textbf{for} the labels in that order: $G_j\mapsto$ \textsc{Gibbs}$(S,j,\xi)$, $U\mapsto$ \textsc{UniformSwap}, each $N_b\mapsto$ \textsc{NeighbourSwap}, $I\mapsto$ \textsc{Independence} %@ v3_engine.h:195-214
\EndProcedure
\Procedure{Gibbs}{$S,j,\xi$}\Comment{inclusion update of predictor $j$}
\State \textbf{if} $j\in S$ \textbf{then} $S^+\gets S$, $S^-\gets S\setminus\{j\}$; \textbf{else if} $\abs S<K$ \textbf{then} $S^+\gets S\cup\{j\}$, $S^-\gets S$; \textbf{else} $o_j\gets0$ %@ v3_engine.h:115-127
\State $a\gets\gamma_\xi(S^+)/\gamma_\xi(S^-)$ and $o_j\gets a/(1+a)$ (evaluated as a logistic function of $\log a$); \textsc{Eval} of the other configuration; if the round is a measured one, add $o_j$ to $\varrho_j$ %@ v3_engine.h:122-128; v3_util.h:30-34
\State draw $u$ (one uniform, also when $o_j\in\{0,1\}$); include $j$ iff $u<o_j$; $S\gets S^+$ if included, else $S^-$ (for $\abs S=K$, $j\notin S$ the model is left as it is) %@ v3_engine.h:129-131
\EndProcedure
\Procedure{UniformSwap}{$S,\xi$}
\State \textbf{if} $\abs S\in\{0,p\}$ \textbf{then} do nothing (no uniform used) %@ v3_engine.h:148
\State $j\gets$ a member of $S$, $l\gets$ a non-member, each uniform ($\lfloor uk\rfloor$ over the increasing lists); $S'\gets(S\setminus\{j\})\cup\{l\}$; accept with probability $\min\{1,\ \gamma_\xi(S')/\gamma_\xi(S)\}=\min\{1,\ \exp[\log\gamma_0(S')-\log\gamma_0(S)+\xi\{\Delta(S')-\Delta(S)\}]\}$; one uniform $u$, accept iff $u<$ that probability %@ v3_engine.h:150-158
\EndProcedure
\Procedure{NeighbourSwap}{$S,\xi$}
\State \textbf{if} $S=\emptyset$ \textbf{then} do nothing; $j\gets$ a uniform member of $S$; $\mathcal C\gets\mathcal N(j)\setminus S$; \textbf{if} $\mathcal C=\emptyset$ \textbf{then} do nothing (no score, no further uniform) %@ v3_engine.h:164-170
\State $l\gets$ uniform in $\mathcal C$ (increasing list); $S'\gets(S\setminus\{j\})\cup\{l\}$; accept with probability $\min\Big\{1,\ \dfrac{\gamma_\xi(S')}{\gamma_\xi(S)}\cdot\dfrac{\abs{\mathcal N(j)\setminus S}}{\abs{\mathcal N(l)\setminus S'}}\Big\}$ (the reverse proposal picks $j$ among $\mathcal N(l)\setminus S'\ni j$) %@ v3_engine.h:171-179
\EndProcedure
\Procedure{Independence}{$S,\xi$}
\State $S'\gets$ \textsc{DrawStart} (an exact draw from $\pi_0$; always scored); accept with probability $\min\{1,\ \exp[\xi\{\Delta(S')-\Delta(S)\}]\}$ ($=\min\{1,\gamma_\xi(S')\pi_0(S)/(\gamma_\xi(S)\pi_0(S'))\}$); one uniform $u$ is always used %@ v3_engine.h:182-191
\EndProcedure
\end{algorithmic}}
\end{algorithm}
