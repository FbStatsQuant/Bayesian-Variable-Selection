\begin{itemize}
\item \textbf{R}: $\mu=X\theta^*$, $\|\mu\|^2\le Mn$, and $\lambda_{\min}(\hat{R}_{S_0})\ge\phi_0^2>0$ ($\hat{R}=X^\top X/(n-1)$);
\item \textbf{tail cutoff}: $R_m=\sum_{k=\max(m+1,0)}^{K}\binom pk w(k)/w(s_0)$, $m^\sharp_n=\min\{m\ge s_0:R_m\le n^{-(s_0+2)/2}\}$ (deterministic, data-free);

\item \textbf{beta-min}: BM-∞, $\theta^{*2}_{\min}n\phi_0^2/(\sigma^2\log n)\to\infty$; or BM$(c)$ with a fixed $c>c_0(s_0,A)$, meaning $\theta_{\min}^{*2}\ge c\sigma^2\log n/(n\phi_0^2)$ eventually, $c_0=\max\{4C_*,32[s_0(A+1)+2]\}$ ($C_*$ the locked T03 proof constant).
\end{itemize}

\end{theorem}
