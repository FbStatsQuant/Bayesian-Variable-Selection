\section{The method}

\textbf{Target.} Gaussian linear model on standardised data with a $g$/ridge prior $\beta_S\mid\sigma^2\sim N(0,n\sigma^2(X_S^\top X_S+I)^{-1})$
($g=n$), $\sigma^2\sim\mathrm{IG}(0.01,0.01)$, and a Beta-Binomial$(1,p)$ size prior truncated at $K=\min(p,\lfloor n/2\rfloor)$; the score of a model is
$F(S)=\log\varpi(S)+\log m(S)$ and $\pi(S\mid y)\propto e^{F(S)}$ on $\Omega=\{|S|\le K\}$.
% SRC: Theory_work/State_of_Project/sections/02_method.tex:78-102 (statement from Docs/Methodology/METHODOLOGY_v3.md:97-108); cap K: 02_method.tex:132
The sampler is a sequential Monte Carlo sampler over model space that starts from an \emph{exactly sampled} block-product law and tempers towards $\pi$:
\begin{enumerate}
\item \textbf{Blocks and exact start law $\pi_0$.} Cluster columns (complete linkage on $1-|r_{jl}|$) into blocks of size $\le B$ ($B\le16$); score every sub-model of every block against the full $y$;
 $\pi_0(S)\propto\varpi(S)\prod_b m(S\cap C_b)/m(\varnothing)$. Its normaliser $\bar Z_0$ is computed exactly (convolution of block size profiles) and $\pi_0$ is sampled exactly and iid.
 % SRC: 02_method.tex:185-203 (METHODOLOGY_v3.md:164-177); B rule 02_method.tex:149 and :205
\item \textbf{Bridge.} $\gamma_\varphi(S)=\gamma_0(S)\,e^{\varphi d(S)}$, $d(S)=\log\mathrm{BF}(S)-\sum_b\log r_b(S\cap C_b)$, $\varphi:0\to1$; $\gamma_1\propto\pi$.
 % SRC: 02_method.tex:200-201
\item \textbf{Grid calibration.} A pilot SMC chooses temperatures by population ESS $\ge0.90$; the final grid is rebuilt from the scored models with a restricted-ESS criterion (escalation tier if diagnostics fail).
 % SRC: 02_method.tex:304-327 (METHODOLOGY_v3.md:205-227); ESS 0.90 at :309
\item \textbf{Islands.} $R$ independent islands ($R=4$ in the study), each: reweight by $e^{(\varphi_t-\varphi_{t-1})d}$, resample (multinomial, every stage), move (a round of $p$ Gibbs coordinate updates, swaps, and one independence proposal from $\pi_0$; each step $\gamma_\varphi$-reversible). Evidence
 $\hat Z_r=m(\varnothing)\,\bar Z_0\prod_t N^{-1}\sum_i e^{(\varphi_t-\varphi_{t-1})d(S_i^{(t-1)})}$.
 % SRC: R=4 02_method.tex:155; moves 02_method.tex:264-281; resampling 02_method.tex:306; Zhat_r 02_method.tex:422
\item \textbf{Outputs.} $\hat Z$ (mean of $\hat Z_r$); Rao--Blackwell PIPs (conditional inclusion probabilities recorded before each Gibbs draw, averaged over islands); MPM $=\{\mathrm{PIP}>\tfrac12\}$;
 \textbf{HPM = best model scored anywhere} (a ledger of every model scored: tables, pilot, islands, independence proposals, searches).
 % SRC: Theory_work/Wave3/W3-SAMP/REPORT.md:43-53 (Lemma CC :43-51, Lemma PC :52-53); RB PIP recording 02_method.tex:270-271
\item \textbf{Final local search with certificate.} A greedy search (also run from the mode of $\pi_0$) ends with a flag \texttt{local\_certified}: no single deletion, addition or swap of the reported model scores higher.
 % SRC: Theory_work/Wave3/W3-SCR/REPORT.md:13-18 (code check CC1-CC6)
\end{enumerate}

\textbf{What is unusual.} (i) SMC over model space with an \emph{exact} start law, so the bridge starts at a law we can sample and normalise exactly; (ii) each island's $\hat Z_r$ is unbiased for $Z$ (Thm~S1 below); (iii) the reported model is a \emph{search result} over everything scored, returned with a local-optimality certificate, and the theory is about that reported object, not only about the posterior.

\subsection*{A. Posterior concentration and selection}
\textbf{Thm P1 (concentration).} If $\|X\theta^*\|^2\le Mn$, sparse-eigenvalue floors hold on the sets $S_0\cup J$ with $|J|\le j_0$ (fixed) and $|J|\le m^\sharp_n=O(\log n)$, and beta-min $\theta^{*2}_{\min}\ge c\,\sigma^2\log n/(n\phi_0^2)$ ($c>c_0$), then
$\delta_n=O_P(r_n)$; exact HPM $=$ MPM $=S_0$ w.p.\ $\to1$; $\max_j|\mathrm{PIP}_j-\mathbf 1\{j\in S_0\}|\le\delta_n$; without beta-min, no mass on strict supersets of $S_0$. Random designs via an interface (e.g.\ Gaussian rows with a sparse eigenvalue floor; pervasive factors allowed for this result).
\st{proved, checked}

\textbf{Design-free size tail.} With no eigenvalue condition: models larger than $m^\sharp_n=O(\log n)$ carry mass $O_P(n^{-1/2})$, uniformly over all $X$ with $\|X\theta^*\|^2\le nM$. \st{proved, checked}

\textbf{Rate lower bound.} For every design and data set with $1\le s_0<p$, $s_0+1\le K$: $\delta_n\ge 1/(1+\sqrt{n+1})$; so $r_n$ is sharp up to $\sqrt{\log n}$.
\st{proved, checked}

(b) $E\,\delta_n\le C r_n$ uniformly over designs. (c) In-sample Hellinger contraction $\bar h_n^2=O(\log n/n)$ without beta-min.

\subsection*{C. The reported model}
\textbf{Thm C1 (certified screening; design-free).} Fixed design, Gaussian errors; no condition on the start, blocks, leakage or calibration. If the sparse visibility $\Gamma_n(q)$ (noiseless gain of the best missing truth, minimised over models of size $\le q$ that miss a truth) satisfies $\sqrt{\Gamma_n(q_n)}\ge(1+\delta)[\tau_n\sqrt{2\log p+\log n}+\sigma\sqrt{2q_n\log p}]$, $\tau_n^2=\sigma^2+\|X\theta^*\|^2/(n-1)$, then every locally optimal model of size $\le q$ contains $S_0$; hence $P(\mathcal O\cap\{\text{certified}\}\cap\{|\Shat|\le q_n\}\cap\{S_0\not\subseteq\Shat\})\to0$. Sufficient: a sparse eigenvalue floor on sets of size $q_n+s_0$ with $(q_n+1)\log p+\log n\le c_1n$. With $s_0$ bounded and a near-orthogonality floor $\phi\ge 2A/(2A+1)+\varepsilon$, a sparse certified model \emph{equals} $S_0$. The constant 1 in the threshold cannot be lowered.
\st{proved, checked}
% SRC: Theory_work/Wave3/W3-SCR/REPORT.md:20-47 (SCR-A :22, A-infinity :26, A1 :27-29, SCR-C :41-43, SCR-N :45-46); status table :56-66; RESULT :7-9; Gamma_n, tau_n: Theory_work/Wave3/W3-SCR/STATEMENT.md:58-59

\textbf{Thm C2 (reported model under visibility and bounded leakage).} Fixed design, Thm P1's conditions, eigenvalue floors on all sets of size $\le16$, and a deterministic $Q\supseteq S_0$ with $|Q|\le(\log n)^\kappa$. Write $\delta^2_{U,l}=\|(P_{U\cup\{l\}}-P_U)X\theta^*\|^2$ (gain in explained mean when $l$ joins $U$). 
(V) $\delta^2_{U,j}\ge M_n\log n$, $M_n\to\infty$, for every truth $j$ and every $U\not\ni j$, $|U|\le15$ (no truth is masked by $\le15$ variables); (L) $\delta^2_{U,l}\le\eta\,\tau_U^2\log n$ for every $l\notin Q$ and such $U$ ($\tau_U^2$ = residual variance left by $U$); (N) a lattice condition, automatic when $|Q\setminus S_0|\le(1-\delta_0)\log n$.
Then $P(\mathcal O\cap\{\Shat\ne S_0\})\to0$ for every seed, number of particles and islands, and calibration outcome. Gaussian-row version with a localisation clause. Examples: metric-local covariances at any $p\le n^A$ (including $p\gg n$), AR(1).
\st{proved, sign-off pending}; dropping (N) \st{open}.

\subsection*{D. Sampler outputs (the method as coded)}
\textbf{Thm S1 (unbiasedness).} For every island, every measured round and every bounded $h$: $E[\hat Z_r\,N^{-1}\textstyle\sum_i h(S_i)\mid\text{calibration}]=Z\,\pi(h)$; in particular $E[\hat Z_r\mid\text{calibration}]=Z$. No condition on $X$.
\st{proved, checked}
% SRC: Theory_work/Wave1C/FINAL/sections/s4_sampler.tex:10-15 (Lemma N1; status :21 "W1-N1 PROVED-CHECKED")

\textbf{Thm S2 (draw discovery).} $P(\text{no draw of an island equals }S_0)=(1-\pi_0(S_0))^{D_r}$ exactly ($D_r$ = draws per island), so $P(\mathcal O\cap\{\Shat\ne S_0\})\le P(\text{exact HPM}\ne S_0)+E(1-\pi_0(S_0))^{D_r}$; it suffices that $\max(n,p)\log_2(n+1)\,\pi_0(S_0)\to\infty$, with no condition on mixing or the bridge.

\textbf{Thm S3 (concentrated bridge).} For any grid and particle number, $E|\hat Z_r/Z-1|\le 2\bar\delta$, $\bar\delta$ = the larger of the off-$S_0$ odds under $\pi$ and $\pi_0$; with concentration of $\pi$ and $\pi_0$: $\hat Z/Z\to1$ and consistent PIPs, MPM and HPM.

\textbf{Thm S4 (terminal repair).} The $\varphi=1$ move kernel is minorised by $\pi/M$, $M=\max\pi/\pi_0$; with posterior concentration and $L/M\to\infty$ ($L$ = burn-in rounds) the PIPs and MPM are consistent with no condition on $\hat Z$, the bridge or $N$.
\st{S2--S4: proved, sign-off pending}; whether the non-collapse condition in the PIP theorem is necessary \st{open}.
