\subsection{The route: M5(0), operational closure, LS0}\label{sec:route}
By M5(0) (§\ref{sec:setting}) it suffices to show $\EE_0[1-\pi(S_0)]\to0$ (layer (i), §\ref{sec:posterior}) and $D_n\to0$. LS0 reads no uniforms, so the models it scores are a deterministic function of the data; under the named premises the \textbf{operational closure} [OP] gives
\[
\bar{\Ocal}\cap\{S_0\notin\bar{\mathcal S}^{+}\}\subseteq\{\text{LS0 from }\tilde{S}_0\text{ does not score }S_0\},\qquad\text{so}\qquad D_n\le\PP(\text{LS0 misses }S_0)
\]
(\st{PROVED (by this seat, unchecked)} inside Theorem GEN \st{PROVED-CHECKED}; XF VERIFIED as a theorem under (OP) in W1-PXN) \srcq{W1/W1-GEN/STATEMENT.md:169-171}{Operational closure [OP]}; \srcq{W1/W1-PXN/REPORT.md:43}{operational closure OP-LS0 + GUARD}. Every theorem in this section is therefore a statement about two data events: the start mode lies in a region from which the greedy search reaches $S_0$, and the search does reach it within its move cap. \textbf{None of them is a delivery claim}; the liminf form $\liminf\PP(\Ocal\cap\{\Shat=S_0\})=1-\limsup\PP(U_n)$ is algebra from $\PP(\Ocal\cap\{\Shat\ne S_0\})\to0$.

\subsection{Theorem PXN: masked correlated design, p ≫ n (LH-iid)}\label{sec:pxn}
\textbf{Family LH-iid.} Fixed $g\ge2$, $c\ge1$, $v\ge0$, $\rho\in(0,1)$, links $0<|\psi_i|<\rho$, nonzero $\beta_i,\beta^{\rm vis}_h$, $\sigma^2>0$, $A\ge1$. $p_n=G_ng$, $18\le p_n\le n^A$, $p_n\to\infty$. Rows iid $N(0,\Sigma)$: groups of size $g$ equicorrelated at $\rho$; $c$ linked group pairs with rank-one cross-correlation through the hubs ($\mathrm{corr}(j_a,j_b)=\psi$); $v$ visible groups; the rest null. Truth: $\theta_{j_{a_i}}=\beta_i$, $\theta_{j_{b_i}}=-\psi_i\beta_i$, visible hubs $\beta^{\rm vis}_h$; $s_0=2c+v$. The $b$-hubs are \emph{masked} (zero population covariance of their block with the signal); reference $S^\circ=S_0\setminus\mathcal M$. \textbf{(SIZE)} $g\le B_n<2g$ eventually ($9\le g\le16$ asymptotically) \srcq{W1C/C-PACK/item5_PXN_consolidated/00_statement_and_conventions.md:8-30}{g ≤ B_n < 2g}. Method: v3 + LS0 (one production local search from the exact start mode, centre emitted, cap $\ell_n$), under (OP) = EX-A, EX-S, CAL, SR, GUARD, OP-LS0, with the \textbf{exact-branch convention} (blocks, tables, $\pi_0$, $\tilde{S}_0$ and ``LS0 of $\bar{R}^{+}$'' are the deterministic data functions on every branch).

Write $s=|S^\circ|=c+v$, $\bar{\psi}=\max_i|\psi_i|$, $q_R=\sqrt{(1-\bar{\psi}^2)/(1-\rho^2)}$, $t^*=(q_R-1)/(q_R\rho-\bar{\psi})\in(\bar{\psi},\rho)$ and $I(t;r)=-\frac12\log[(1-r^2)(1-t^2)/(1-rt)^2]$.
\begin{theorem}[PXN]\label{thm:PXN}
Under LH-iid with (SIZE), $18\le p_n\le n^A$, $p_n\to\infty$, $s_0$ fixed, and (OP), as $n\to\infty$:
\begin{itemize}
\item[(R)] $\PP(\text{blocks}\ne\text{groups})\to0$, summable, and for every fixed $C>0$, $\le p^{-C}$ for $n\ge\max\{n_1,n_0(A,C,\rho,\bar{\psi})\}$ ($n_1$ the SIZE onset); explicitly $\le\frac{2}{\sqrt\pi(1-\rho)}p^2\sqrt n\,e^{-(n-4)I^*}$ for $n\ge5$ with SIZE at $n$, $I^*=I(t^*;\bar{\psi})>0$.
\item[(S)] $1-\pi_0(S^\circ)=O_P(\sqrt{\log n/n})$; $\pi_0(S_0)=p^{-c}n^{-c/2}e^{O_P(1)}\to0$, more precisely $\log\pi_0(S_0)+c\log p+\frac c2\log n\to_d\sum_{i\le c}\log\frac{s+i}2+\frac12\chi^2_c$ (the masked regime).
\item[(E)] $\PP(\tilde{S}_0=S^\circ)\to1$.
\item[(T)] $\PP(\text{LS0 scores }S_0)\to1$ and $\PP(\text{LS0 returns }S_0\text{ with \texttt{local\_certified}})\to1$.
\item[(P)] $1-\pi(S_0)=O_P(\sqrt{\log n/n})$ under the joint law of $(X^r,\varepsilon)$.
\item[(C)] For every $n$: $\PP(\Ocal\cap\{\Shat\ne S_0\})\le2\EE_0[1-\pi(S_0)]+D_n$, $D_n\le\PP(\text{LS0 misses }S_0)\to0$; hence $\PP(\Ocal\cap\{\Shat\ne S_0\})\to0$ and $\liminf\PP(\Ocal\cap\{\Shat=S_0\})=1-\limsup\PP(U_n)$.
\item[(K)] LS0 costs $\le1+(\ell_n+1)\max_{k\le K}[p+k(p-k)]=O(p^2\log(n+p))$ scores; on the joint event $\mathcal T_n$ (lattice arrival, (L), $\ell_n\ge c$) it uses $c+1$ scans.
\end{itemize}
\end{theorem}
\textbf{Corrected centring remark.} The finite-$n$ centring added after the W1-PXN council claimed a remainder $O_P(\sqrt{\log n/n})$; this fails for slowly growing $p$. The correct remainder is $O_P(\sqrt{\log n/n}+1/p)$ (prior correction $d_p=\sum_{i\le c}\log[2p/(2p-s-i)]=O(1/p)$); the limit law in (S) and the headline are unaffected, and no rate of convergence to the $\chi^2$ law is asserted \srcq{W1C/C-PACK/item5_PXN_consolidated/03_start.md:111}{O_P(√(log n/n) + 1/p)}.
