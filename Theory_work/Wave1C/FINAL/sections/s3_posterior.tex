\subsection{Posterior consistency with rate (S-T, T03♯)}
\begin{theorem}[S-T / T03♯; card TH09]\label{thm:ST}
Fixed-design triangular array (model G), fixed $s_0\ge1$, $p\le n^A$ for a fixed $A$, $s_0\le K$ eventually. Assume
\begin{itemize}
\item \textbf{R}: $\mu=X\theta^*$, $\|\mu\|^2\le Mn$, and $\lambda_{\min}(\hat{R}_{S_0})\ge\phi_0^2>0$ ($\hat{R}=X^\top X/(n-1)$);
\item \textbf{tail cutoff}: $R_m=\sum_{k=\max(m+1,0)}^{K}\binom pk w(k)/w(s_0)$, $m^\sharp_n=\min\{m\ge s_0:R_m\le n^{-(s_0+2)/2}\}$ (deterministic, data-free);
\item \textbf{SE$_\lambda(j_0)$}, $j_0=\lceil(2A+1)s_0\rceil+1$: every $S_0\cup J$, $|J|\le j_0$, has normalised minimum eigenvalue $\ge\phi_0^2$; \textbf{SE$_w(m^\sharp_n;g_0)$}: every $S_0\cup J$, $|J|\le m^\sharp_n$, has unnormalised minimum eigenvalue $\ge g_0>0$;
\item \textbf{beta-min}: BM-∞, $\theta^{*2}_{\min}n\phi_0^2/(\sigma^2\log n)\to\infty$; or BM$(c)$ with a fixed $c>c_0(s_0,A)$, meaning $\theta_{\min}^{*2}\ge c\sigma^2\log n/(n\phi_0^2)$ eventually, $c_0=\max\{4C_*,32[s_0(A+1)+2]\}$ ($C_*$ the locked T03 proof constant).
\end{itemize}
Then $1-\pi(S_0)=O_P(\sqrt{\log n/n})$; the exact HPM and MPM are consistent. Without beta-min, for $J_n(c)=\{j\in S_0:\theta_j^{*2}\ge c\sigma^2\log n/(n\phi_0^2)\}$, $c>c_0$: $\pi(S\not\supseteq J_n(c))=O_P(\sqrt{\log n/n})$.
\end{theorem}

\subsection{Sure screening, HPM, MPM, thresholded models (C-PACK item 4)}
\begin{lemma}[SL0, deterministic; every data set, every $n$]\label{lem:SL0}
Let $\delta:=1-\pi(S_0)$, with $\pi(S_0):=0$ if $s_0>K$; every maximum of nonnegative coordinate errors over an empty set is defined as zero. Then: (i) $|\mathrm{PIP}_j-1\{j\in S_0\}|\le\delta$ for every $j$, hence $\max_{j\in S_0}(1-\mathrm{PIP}_j)+\max_{j\notin S_0}\mathrm{PIP}_j\le2\delta$; (ii) $|\pi(A\subseteq S)-1\{A\subseteq S_0\}|\le\delta$ for every $A$, and if $S_0\in\Omega$, $|\pi(h)-h(S_0)|\le\delta\operatorname{osc}(h)$ and $\sup_{h:\Omega\to[0,1]}|\pi(h)-h(S_0)|=\delta$; (iii) $\delta<\min(q,1-q)\Rightarrow\xi_q=S_0$, in particular $\delta<\frac12\Rightarrow\mathrm{MPM}=S_0$; (iv) $\delta<\frac12\Rightarrow S_0$ is the unique maximiser of $\pi$; (v) $\pi(|S|=s_0)\ge1-\delta$. (vi) When $1\le s_0\le K$ and $s_0<p$, the constant 2 in (i) is sharp over arbitrary laws on $\Omega$ (two-point law); attainment by a D0 posterior is \emph{not} claimed.
\end{lemma}
\begin{corollary}[SL1]\label{cor:SL1}
If $\delta_n=O_P(r_n)$, $r_n\to0$, under some law: (a) $\max_j|\mathrm{PIP}_j-1\{j\in S_0\}|=O_P(r_n)$ and $\EE\max_j|\cdot|\le\EE\delta_n\to0$ (no expectation rate); (b) for every deterministic sequence $q_n\in(0,1)$, $\PP(\xi_{q_n}=S_0)\to1$ whenever $r_n=o(\min(q_n,1-q_n))$, in particular for every fixed $q\in(0,1)$, and $\PP(\mathrm{MPM}=S_0)\to1$; (c) $\PP(\mathrm{HPM}=S_0)\ge\PP(\delta_n<\frac12)\to1$, $\PP(\mathrm{HPM}\ne S_0)\le2\EE\delta_n$; (d) $\sup_{h:\Omega\to[0,1]}|\pi(h)-h(S_0)|=O_P(r_n)$; (e) $\pi(|S|=s_0)=1-O_P(r_n)$. Instances with $r_n=\sqrt{\log n/n}$: \textbf{I-a} all T03♯ hypotheses (fixed design); \textbf{I-b} all W1-PXN (P) hypotheses (LH-iid, joint law); \textbf{I-c} GEN (C4) (joint law); \textbf{I-d} LH exact Gram (Thm5-LH keeps \st{SAME-FAMILY-ONLY}).
\end{corollary}
\begin{corollary}[SL1′: no beta-min, large signals only]\label{cor:SL1p}
Retaining T03♯'s other hypotheses, $\max_{j\in J_n(c)}(1-\mathrm{PIP}_j)\le\pi(S\not\supseteq J_n(c))=O_P(\sqrt{\log n/n})$ ($=0$ if $J_n(c)=\varnothing$). No specificity or full-support recovery follows from this clause alone.
\end{corollary}

\subsection{Misspecified candidate sets (C-MIS; Song--Liang Thm 4 / Cor 2 analog)}\label{sec:mis}
\textbf{Setting.} A candidate set $\mathcal C$, $|\mathcal C|=p_{\mathcal C}\le n^A$; the method is \emph{re-run on the candidate data} $(X^r_{\mathcal C},y^r)$ with the shipped score and prior (BB$(1,p_{\mathcal C})$, $K=\min(p_{\mathcal C},\lfloor n/2\rfloor)$) — not obtained by conditioning the full-$p$ posterior. True model $y^r=\alpha\mathbf 1+X^r_{\mathcal C}\beta_{\mathcal C}+\omega+\varepsilon$, $\omega=X^r_{\mathcal M}\beta_{\mathcal M}$ the omitted contribution \srcq{W1C/C-MIS/STATEMENT.md:37-50}{re-run on the candidate data}.

\textbf{(GR) Gaussian residual.} Conditionally on $X^r_{\mathcal C}$, $\omega=a\mathbf 1+X^r_{\mathcal C}\gamma+\xi$ with $\xi\sim N_n(0,\tau^2I_n)$, $(a,\gamma)$ measurable in $X^r_{\mathcal C}$, $\tau^2\ge0$ \textbf{fixed}, $\xi$ independent of $\varepsilon$; the method's uniforms independent of everything. Define $\beta^{\mathcal C}:=\beta_{\mathcal C}+\gamma$, $S_0^{\mathcal C}:=\operatorname{supp}\beta^{\mathcal C}$, $s_0^{\mathcal C}:=|S_0^{\mathcal C}|$, $\sigma^2_{\mathcal C}:=\sigma^2+\tau^2$. Under GR the candidate-only problem \emph{is} a well-specified Gaussian linear model with truth $\beta^{\mathcal C}$ and noise $\sigma^2_{\mathcal C}$ \srcq{W1C/C-MIS/STATEMENT.md:54-58}{(GR).}.

\textbf{(H-MIS)} = the T03♯ hypotheses for the candidate problem with \[(p,S_0,s_0,\theta^*,\mu,\sigma^2)\mapsto(p_{\mathcal C},S_0^{\mathcal C},s_0^{\mathcal C},\theta^{\mathcal C},\mu^{\mathcal C},\sigma^2_{\mathcal C}),\] in particular $s_0^{\mathcal C}\ge1$ fixed and beta-min relative to $\sigma^2_{\mathcal C}$.

\begin{theorem}[MIS]\label{thm:MIS}
(i) \emph{Fixed candidate design}, GR and (H-MIS): $1-\pi_{\mathcal C}(S_0^{\mathcal C}\mid y)=O_P(\sqrt{\log n/n})$; $\max_j|\mathrm{PIP}_j-1\{j\in S_0^{\mathcal C}\}|\le1-\pi_{\mathcal C}(S_0^{\mathcal C}\mid y)$; $\PP(\mathrm{MPM}_{\mathcal C}\ne S_0^{\mathcal C})\to0$; for the exact-posterior HPM $S^\dagger_{\mathcal C}=\arg\max\pi_{\mathcal C}$, $\PP(S^\dagger_{\mathcal C}\ne S_0^{\mathcal C})\le2\EE[1-\pi_{\mathcal C}(S_0^{\mathcal C}\mid y)]\to0$; without beta-min, $\pi_{\mathcal C}(S\not\supseteq J_n^{\mathcal C}(c)\mid y)=O_P(\sqrt{\log n/n})$ for fixed $c>c_0$. Consequences: a \emph{proxy} ($j\in S_0^{\mathcal C}\setminus S_0$) has $\mathrm{PIP}_j\to1$; a \emph{cancelled} true candidate ($\beta_j+\gamma_j=0$) has $\mathrm{PIP}_j\to0$. \textbf{Corollary N}: every $T\ne S_0^{\mathcal C}$ has $\pi_{\mathcal C}(T\mid y)\le1-\pi_{\mathcal C}(S_0^{\mathcal C}\mid y)$; in particular the retained truth $S_0\cap\mathcal C$ and the sample-projection support lose all mass whenever they differ from $S_0^{\mathcal C}$.\\
(ii) \emph{Random candidate design, envelope form}: the same conclusions under the joint law, if on measurable design sets $\mathcal A_n$ with $\PP(\mathcal A_n)\to1$ every $x\in\mathcal A_n$ satisfies the finite-$n$ inequalities of (H-MIS) with fixed constants, $s_0^{\mathcal C}(x)=s\ge1$ and the beta-min envelope (diagonal argument, Lemma DR); (ii′) without beta-min, the inclusion clause.\\
(iii) \emph{Jointly Gaussian rows} ($\mathcal M:=S_0\setminus\mathcal C$, rows iid $N(m,\Sigma)$, $\Sigma_{\mathcal C\mathcal C}\succ0$): at each $n$ GR holds with $\gamma=\Sigma_{\mathcal C\mathcal C}^{-1}\Sigma_{\mathcal C\mathcal M}\beta_{\mathcal M}$, $\tau^2=\beta_{\mathcal M}^\top(\Sigma_{\mathcal M\mathcal M}-\Sigma_{\mathcal M\mathcal C}\Sigma_{\mathcal C\mathcal C}^{-1}\Sigma_{\mathcal C\mathcal M})\beta_{\mathcal M}$; $\beta^{\mathcal C}$ is the population slope with intercept (= Song--Liang's $\beta_0$ and $S_0^{\mathcal C}=\tilde{s}$ for centred rows). MIS (i)--(ii) apply only when in addition $\tau_n^2\equiv\tau^2$ is fixed. \textbf{Neighbour rule} (Prop.~P(d)): with one active omitted variable $k$ and $\Theta$ the precision of the marginal vector $(x_{\mathcal C},x_k)$, a candidate that is null in the full model belongs to $S_0^{\mathcal C}$ iff $\Theta_{jk}\ne0$ — partial, not marginal, correlation decides (AR(1): the two neighbours; a one-factor design can make $s_0^{\mathcal C}$ grow, outside the theorem). \textbf{Lemma GD}: (a) for fixed $p_{\mathcal C}$, $m,\Sigma,\beta$ fixed, $\beta^{\mathcal C}\ne0$, (ii) applies and the inclusion clause recovers Song--Liang's Cor.~2 conclusion in the jointly Gaussian case, now with a rate; (b) growing $p_{\mathcal C}\le n^A$ under stated population restricted-eigenvalue, variance, energy and projected beta-min assumptions — projection sparsity is \emph{assumed}.
\end{theorem}
\begin{theorem}[F: fixed design, exact sparse projection]\label{thm:F}
Put $N=n-1$; let $U^\top U=I_N$, $UU^\top=C$, $\mu_c=U^\top X^r\beta$ (the full mean in contrast coordinates) and $X_{c,\mathcal C}=U^\top X_{\mathcal C}$. Here R$^{\mathcal C}$ means $\|\mu_c\|^2\le Mn$ and $\lambda_{\min}(\hat{R}_{S_0^{\mathcal C}})\ge\phi_0^2$ for fixed constants; the energy bound is on the full mean. Assume (EP): $r:=\mu_c-X_{c,\mathcal C}\theta^{\mathcal C}$ is orthogonal to every candidate, $\operatorname{supp}\theta^{\mathcal C}=S_0^{\mathcal C}$; set $\Delta_n=\|r\|^2/N$, $\sigma^2_{\mathcal C,n}=\sigma^2+\Delta_n$. Under $s_0^{\mathcal C}\ge1$ fixed, $p_{\mathcal C}\le n^A$, G, R$^{\mathcal C}$, SE$_\lambda(j_0)$, SE$_w(m_n;g_0)$ anchored at $S_0^{\mathcal C}$ (cutoff $s_0^{\mathcal C}\le m_n\le\min(K,C\log n)$), BM-∞ or BM$(c>c_0)$ relative to $\sigma^2_{\mathcal C,n}$, and the size tail \textbf{TAIL$^{\mathcal C}(m_n)$}: $\pi_{\mathcal C}(|S|>m_n\mid y)=O_P(\sqrt{\log n/n})$ — the conclusions of T03♯ hold for $S_0^{\mathcal C}$. TAIL$^{\mathcal C}$ may be removed (a) for bounded $p_{\mathcal C}$, taking $m_n=m_n^\sharp=K$ eventually (Cor.~F(a)); or (b) if, for a fixed $\kappa\in(0,1)$, (TL) $\sigma^2/[\Delta_n+\sigma^2(1-d_n/N)]\le\frac34(1-\kappa)$ eventually, $d_n=\operatorname{rank}X_{c,\mathcal C}$. In (b), Lemma T supplies the deterministic $m_n^T=s+\lceil(s(A+\frac12)+1)\log n/\log(1/\rho)\rceil$ ($s=s_0^{\mathcal C}$, $\rho=\frac12(\frac14+\frac{3\kappa}8)^{-1/2}$) with $\pi_{\mathcal C}(|S|>m_n^T\mid y)=O_P((\log n)^{s}/n)$; take $m_n=\min(K,\max(m_n^\sharp,m_n^T))$ and require SE$_w(m_n;g_0)$ at this cutoff; all other hypotheses of Theorem F remain (Cor.~F(b)) \srcq{W1C/C-MIS/lemmas/L4_theoremF.md:158-162}{Lemma T.}. \textbf{Lemma Z} (null projection, fixed candidate design): if $s_0^{\mathcal C}=0$, $p_{\mathcal C}$ is bounded, and $\lambda_{\min}(X_{\mathcal C}^{\top}X_{\mathcal C})\ge g_0>0$ eventually, then under GR, or under EP with $\theta^{\mathcal C}=0$, $1-\pi_{\mathcal C}(\varnothing\mid y)=O_P(n^{-1/2})$. The general growing-$p$ null case is not claimed.
\end{theorem}
