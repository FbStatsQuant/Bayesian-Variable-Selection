{\Large\bfseries\sffamily The theorems we aim at}\\[2pt]
{\footnotesize Throughout: $p\le n^{A}$ (any fixed $A$, $p\gg n$ allowed), $s_0$ fixed, Gaussian noise, SMC-BVS as coded on
\texttt{main}, all results asymptotic, conditions on $(X,\theta^*)$ only. Numbers match the status table on the next page.}

\begin{enumerate}[leftmargin=2em,itemsep=4pt,label=\textbf{\arabic*.}]
\item \textbf{Posterior concentration.} $1-\pi(\Szero\mid y)=\OP(\sqrt{\log n/n})$ and HPM $=$ MPM $=\Szero$ w.p.\,$\to1$, for fixed and random Gaussian designs. \hfill{\footnotesize W3-POST, W2-APP}
\item \textbf{Posterior refinements.} $\E_{\theta^*}[1-\pi(\Szero\mid y)]\le C\sqrt{\log n/n}$ (expectation over the noise, $y=\alpha^*\mathbf 1+X\beta^*+\varepsilon$, $\varepsilon\sim N_n(0,\sigma_*^2I)$, design fixed), with $C$ uniform over designs; Hellinger contraction of the posterior; no eigenvalue condition away from $\Szero$. \hfill{\footnotesize W3-POST2}
\item \textbf{Rate optimality.} On every design, no rate $o(n^{-1/2})$ is possible; the $\sqrt{\log}$ factor is attained; $\pi(\Szero)\to1$ requires $s_0=o(\sqrt n)$. \hfill{\footnotesize W3-PSHARP}
\item \textbf{Misspecification.} With omitted variables and any residual variance, the posterior concentrates on the projected model at the same rate. \hfill{\footnotesize W3-MISR}
\item \textbf{Start-mass rate.} The start law gives $\pi_0(\Szero)\ge1-n^{-c}$ for every $c<\tfrac12$, and mass $1-n^{-c}$ on the leakage lattice for $|Q|\le n^{1/2-\epsilon}$; the exponent $\tfrac12$ is optimal. \hfill{\footnotesize W3-START, START2}
\item \textbf{Search.} LS0 started from the start mode ends at $\Szero$. \hfill{\footnotesize W1-GEN, C-GROW}
\item \textbf{Reported model (main theorem).} Under visibility (V), bounded leakage (L) with polylogarithmic $Q$ and a lattice condition (N): $\Prob(\Out\cap\{\Shat\ne\Szero\})\to0$ for every seed and calibration outcome. \hfill{\footnotesize W3-RM}
\item \textbf{Reported model, polynomial leakage.} The same with $|Q|\le n^{b}$, $b<\tfrac12$, under a width condition (W). \hfill{\footnotesize W3-RM2}
\item \textbf{Sampler outputs as coded.} $\hat Z$ is unbiased; the coded PIPs, MPM and HPM are consistent; $\hat Z/Z\to1$ when the start concentrates. \hfill{\footnotesize W3-SAMP, W1-N1}
\item \textbf{Delivery.} The method always returns output, and $\Prob(\Shat=\Szero)\to1$. \hfill{\footnotesize C-DLV, W2-UNI}
\item \textbf{Certified screening.} Every certified model of size $\le q_n$ contains $\Szero$, with no start or leakage condition; exact recovery under a near-orthogonality floor. \hfill{\footnotesize W3-SCR}
\item \textbf{Strong factors at $p\gg n$ (D1).} The full method reports $\Szero$ on strong-factor designs with $p\gg n$, or else a stated limit. \hfill{\footnotesize W3-P6 $\to$ theory}
\item \textbf{Reported model without the lattice condition.} Theorems 7--8 with (N)/(W) removed, via control of the LS0 path. \hfill{\footnotesize W3-PATH}
\item \textbf{Masked truths at growing $Q$.} One theorem that covers both the general-design and masked-hub results. \hfill{\footnotesize ticket T7}
\item \textbf{RM2 riders.} The core reduction holds on any down-closed sub-lattice; the theorem holds for constant-level signals. \hfill{\footnotesize T2, T4}
\item \textbf{Parallel SMC.} An output-identical parallel engine inherits every guarantee above. \hfill{\footnotesize W3-SCALE}
\end{enumerate}

\section{The main results}
Throughout: $s_0=|\Szero|$ fixed, $2\le p\le n^{A}$, Gaussian noise, $K$ the method's model-size cap,
$\rn=\sqrt{\log n/n}$, $\pis$ the coded posterior, $\pi_0$ the start law, $\Out$ the event that a run produces output,
$\Shat$ the reported model. ``As coded'' means the method on \texttt{main} under the named implementation premises
(EX-A exact arithmetic, OP-LS0 shipped configuration, complete scans). These are part of every reported-model
statement.

\subsection*{A. The posterior}

\begin{res}[Posterior concentration; \cid{W3-POST:Thm-1} $=$ \cid{SOP:TH09} (T03$\sharp$)]
Under (R1), fixed design, energy (E), sparse eigenvalue floors (F$_\lambda$), (F$_w$) and beta-min (BM):
\[1-\pis=\OP(\rn),\qquad \Prob(\mathrm{HPM}=\mathrm{MPM}=\Szero)\to1,\]
with thresholded-model, PIP and size consequences, and a random-design interface (item 6). Without (BM), signals
above the detection level are still included.
\end{res}
\statusline{Council W3-POST C3: Astra 8.8 / Opus 8.7 (C1 7.5/8.4 and C2 8.3/8.8 failed and were fixed); SF+XF verified. Random Gaussian rows: \cid{W2-APP:POST} (council 9.1/9.1).}
\begin{paperbox}The foundation every other result consumes. It is the Song--Liang-type consistency the paper
promises, stated design-agnostically. Result~3 shows the rate is optimal.\end{paperbox}

\begin{res}[Posterior refinements; \cid{W3-POST2:A1, C, D, E}]
(A1) The conclusions of Result~1 hold with (F$_w$) replaced by a weaker overfit-size condition (OS).
(C) \emph{Expectation rate, uniform over designs:} $\E[1-\pis]\le C\rn$, where $\E$ is over the noise under the true model G, $y=\alpha^*\mathbf 1+X\beta^*+\varepsilon$, $\varepsilon\sim N_n(0,\sigma_*^2 I)$, with $X$ and $\theta^*$ fixed. $C$ depends only on $(s_0,A,\sigma_*^2,M,\varphi_0,c)$, not on the design. For random designs, $\E\le C\rn+\Prob(\mathcal A_n^c)$ under the joint law of $(X,\varepsilon)$. Hence $\Prob(\mathrm{HPM}\ne\Szero)\le 2\E\delta_n$.
(D) \emph{In-sample Hellinger contraction} of the posterior on $(S,\theta,v)$: rate $M_n/\sqrt n$ given $\delta_n\to0$; squared
rate $\log n/n$ under (OS) without beta-min.
(E) Items 1--4 of Result~1 with \emph{no eigenvalue condition away from $\Szero$}, replaced by an identifiability margin (ID$^M$).
\end{res}
\statusline{Council W3-POST2 C1: Astra 8.6 / Opus 8.6, all gates PASS. The registry label is pending (\S1.1).}
\begin{paperbox}Completes the Song--Liang package (their Thms 1--2: predictive/Hellinger). (E) is the most
design-agnostic form of the foundation; (C) is the input the misspecification theorem uses.\end{paperbox}

\begin{res}[Sharpness / lower bounds; \cid{W3-PSHARP:S, A, B, C}]
(B) For \emph{every} design and data set with $0\le s_0\le K-1$: odds against $\Szero$ are $\ge c(p,s_0)/\sqrt n$ with
$c\ge(s_0+1)/3$. Hence no rate $o(n^{-1/2})$ is possible, $\pi(\Szero)\to1$ needs $s_0=o(\sqrt n)$, and rate $\rn$ needs
$s_0=O(\sqrt{\log n})$.
(A) On witness designs (orthogonal, or spherical with $p\gg n$) $\delta_n\ge(1-o_{\mathbb P}(1))(s_0+1)\sqrt{\log p/(\pi n)}$,
so the $\sqrt{\log}$ factor is attained.
(C) The coefficient/ridge-cost control in the size-tail lemma cannot be dropped (explicit counterexample).
\end{res}
\statusline{PROVED-CHECKED, council R2: Sol 9.2 / Opus 9.3 (4 claims). Implies \cid{W3-POST:Prop-LB}.}
\begin{paperbox}Answers the referee question ``is your rate optimal?'': yes, for $\log p\asymp\log n$. It also
gives the honest sparsity frontier. This is the sharpness gate for the posterior layer.\end{paperbox}

\begin{res}[Misspecification without fixed variance; \cid{W3-MISR:R$'$, MIS(i), MIS(ii), Cor-G, Cor-H}]
(R$'$) The coded posterior is exactly invariant under affine maps of $y$ (code-pinned; 25 rescaled fits,
$|\Delta\mathrm{PIP}|\le1.3\times10^{-10}$). (MIS(i)) For the candidate-only fit with omitted variables (law GR$^c$),
\emph{any residual variance $v_n>0$}:
$\sup\E[1-\pi_C(\Szero^C)]\le C\rn$, with constants free of $v_n$, plus HPM/MPM/PIP rates. (MIS(ii)) Random candidate
designs. (Cor-G) Gaussian-row example. (Cor-H) Hellinger contraction around the projected law.
\end{res}
\statusline{PROVED-CHECKED, council R2: Sol 9.2 / Opus 8.9. R$'$ is unconditional; 6 of 7 are conditional on POST2 C/E/A1, TH09/TH128, C-MIS Prop-P/GD(b), all council-passed (\S1.1).}
\begin{paperbox}The Song--Liang Thm~4 analogue, and stronger than our old \cid{C-MIS} (which fixed $\tau^2$). It is a
candidate to \emph{supersede} C-MIS (decision~2).\end{paperbox}

\subsection*{B. Start and search}

\begin{res}[Start-mass rate; \cid{W3-START} (21 claims) $+$ \cid{W3-START2} (9 claims)]
\emph{Fixed design} (START-F, sharp form): odds against $\Szero$ under $\pi_0$ are
$\OP\!\big(n^{-1/2}L^{\sharp}_n(h)+n^{-1}L_n(\Szero;h)^2\big)+o_{\mathbb P}(n^{-\kappa})$, where $L$ is a leakage functional; the
mass outside the lattice $\{\Szero\subseteq T\subseteq Q\}$ is $\OP(\rho_n(Q;h))$. \emph{Gaussian rows} (START-R(b)): w.p.\,$\to1$,
\begin{gather*}
1-\pi_0(\mathcal L(Q))\le(|Q|+1)\,n^{-1/2}e^{C_R\sqrt{\log n}},\\
Q=\Szero\ \Rightarrow\ \pi_0(\Szero)\ge1-n^{-c}\ \ \forall c<\tfrac12;\qquad
|Q|\le n^{1/2-\epsilon_Q}\ \Rightarrow\ \pi_0(\mathcal L(Q))\ge1-n^{-c}\ \ \forall c<\epsilon_Q.
\end{gather*}
Random-design transfer (TRANSFER). Limits (NEC): the exponent $\tfrac12$ is optimal and the visibility constant $1+2A$ cannot be lowered.
\end{res}
\statusline{PROVED-CHECKED, unconditional. START council R2 Sol 9.2 / Opus 8.9; START2 council R1 Sol 9.1 / Opus 9.1. ``Headline strength'' pending the literature gate.}
\begin{paperbox}New this week. It discharges the start premise (P-START, P-OVL) of the sampler theorems
(Result~9), so the start link of the chain is now proved, not assumed. START2 is a candidate to supersede 8 START
claims (decision~3).\end{paperbox}

\begin{res}[LS0 search; \cid{W1-GEN:Thm-GEN}, \cid{C-GROW:Thm-GROW}]
On general correlated fixed designs with a bounded separated relevance set and a block-level centre certificate:
search failure $D_n\to0$ and $\Prob(\Out\cap\{\Shat\ne\Szero\})\to0$ (GEN). The same holds with a growing relevance set
$|Q|\le C_Q\log n$ under the MR1 cap (GROW).
\end{res}
\statusline{PROVED-CHECKED. GEN: Sol 9.0 / Opus 8.9; GROW: Sol 9.2 / Opus 9.0. Both are special cases of the RM line (Results 7--8).}
\begin{paperbox}Historical route to the main theorem. In v2.0 it becomes a remark or special case of Result~7.\end{paperbox}

\subsection*{C. The reported model (main consistency theorem)}

\textbf{The three conditions in plain words.} \emph{(V) visibility:} no truth can be masked by conditioning on
$\le15$ other variables (block size of the coded partition). \emph{(L) bounded leakage:} there is a deterministic set
$Q\supseteq\Szero$, small, outside which no column gains more than $\eta\sigma^2\log n$ in fit. \emph{(N)/(W)
lattice/width condition:} a deterministic condition on the residual directions of the lattice
$\{\Szero\subseteq T\subseteq Q\}$. It is automatic when $|Q\setminus\Szero|\le(1-\delta_0)\log n$. Dropping it is W3-PATH.

\begin{res}[Reported model $=\Szero$; \cid{W3-RM:RM-0, RM-F, RM-R}]
(RM-0, any design) For deterministic $Q\supseteq\Szero$, $|Q|\le K$, pathwise for every seed, $N_e$, $R'$ and calibration outcome:
\[\Out\cap\{\Shat\ne\Szero\}\subseteq\{\pi(\Szero)\le\tfrac12\}\cup\mathcal K_n^{c}.\]
(RM-F, fixed design) Result~1's conditions $+$ floors (FL) $+$ (V) $+$ (L) with $|Q|\le(\log n)^{\kappa}$ $+$ (N) $\Rightarrow$
$\Prob(\Out\cap\{\Shat\ne\Szero\})\to0$. (RM-R, Gaussian rows, $p\gg n$ allowed) the same under the joint law of
$(X,\varepsilon)$ with (V-R)$=$PTV, localised leakage (L-R) and (D-R). Lemma E4 closes premise e4: \texttt{start\_mode}
returns a maximiser of $\pi_0$ on every path.
\end{res}
\statusline{PROVED-CHECKED, council round 2: Astra 9.8 / Opus 9.1 (round 1 failed on the sharpness record and was fixed).}
\begin{paperbox}\textbf{The paper's headline theorem} about the method as coded. RM-0 is a clean, design-free
reduction: the reported model is wrong only if the posterior is not concentrated, or the search leaves a
deterministic sandwich.\end{paperbox}

\begin{res}[Reported model under polynomial leakage; \cid{W3-RM2:RM2-F, RM2-R} (18 claims)]
(RM2-F) Result~1's conditions, (FL), (V), $|Q|\le n^{b}$ with $b<\beta(\eta,A)=\tfrac12-\tfrac\eta2-\sqrt{2A\eta}$, a
signal floor (BM-W) (implied by $n\theta_{\min}^2\gg|Q|+\log n$) and the width condition (W) $\mathfrak w^0_n\to-\infty$
$\Rightarrow$ one data event of probability $\to1$ on which \emph{every} admissible output run reports $\Szero$; LS0 makes
$\le n^{b}$ deletions and the shipped cap never binds. (RM2-R) Gaussian rows: leak bound
$|\mathrm{Cov}(W_l,W_j\mid W_U)|=o(\rn)$ at every conditioning set $|U|\le15$, localisation without rate, $15\rho+b<\tfrac12$
$\Rightarrow$ the same under the joint law. RM2-F $\Rightarrow$ RM-F and RM2-R $\Rightarrow$ RM-R.
Examples: polynomial decay $\alpha>2$; one truth with $n^b$ weak proxies; an equicorrelated group ($b<3/10$).
\end{res}
\statusline{PROVED-CHECKED, council R1: Astra 9.5 / Opus 8.8. Conditional: F-forms on TH09, R-forms on W2-APP:POST (both council-passed, \S1.1). Ex-PolyR/F are unchecked until RM2's next council round. Boundary probes (LS0 re-implementation): 202/202 end at $\Szero$, $b$ up to $0.75$; LS0 stalls only far outside the leak condition.}
\begin{paperbox}Strictly stronger than Result~7. It is the natural new headline: leakage sets grow polynomially, and
the decay frontier drops from $17/2$ to $2$. Decision~1 is whether RM2 becomes the headline with RM as its corollary.\end{paperbox}

\subsection*{D. Sampler outputs, delivery, screening}

\begin{res}[Sampler outputs as coded; \cid{W1-N1:Thm-N1}, \cid{W3-SAMP:D, LT$\sharp$, C, TR}]
(N1) $\E[\hat Z_rH_r(h)\mid\mathcal C]=Z\,\pi(h)$ for every bounded summary $h$ (any $X$). (D) Draw discovery:
